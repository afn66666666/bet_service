запуск сервера из корня репозитория
cd L:\programming\cpp\bet_service
docker build -t login-service .\Services\login_service
docker run --rm -p 6868:50051 --name login-service -v "${PWD}/secrets/jwt-public.pem:/run/secrets/jwt-public.pem:ro" -v "${PWD}/secrets/jwt-private.pem:/run/secrets/jwt-private.pem:ro" -e JWT_PUBLIC_KEY_PATH=/run/secrets/jwt-public.pem -e JWT_PRIVATE_KEY_PATH=/run/secrets/jwt-private.pem login-service
